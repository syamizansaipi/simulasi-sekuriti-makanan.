    <script>
        // 1. Math Equations Definitions
        // f(x) = -0.005x³ + 0.18x² + 0.2x
        function calculateYield(x) {
            return (-0.005 * Math.pow(x, 3)) + (0.18 * Math.pow(x, 2)) + (0.2 * x);
        }

        // f'(x) = -0.015x² + 0.36x + 0.2
        function calculateDerivative(x) {
            return (-0.015 * Math.pow(x, 2)) + (0.36 * x) + 0.2;
        }

        // 2. Generate Dataset for Years 0 to 30
        const yearsData = [];
        const yieldData = [];
        for (let x = 0; x <= 30; x++) {
            yearsData.push(x);
            yieldData.push(parseFloat(calculateYield(x).toFixed(2)));
        }

        // 3. Initialize Chart.js with fluid animations
        const ctx = document.getElementById('yieldChart').getContext('2d');
        
        // Gradient fill for graph line
        const chartGradient = ctx.createLinearGradient(0, 0, 0, 400);
        chartGradient.addColorStop(0, 'rgba(16, 185, 129, 0.4)');
        chartGradient.addColorStop(1, 'rgba(16, 185, 129, 0.0)');

        const yieldChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: yearsData,
                datasets: [{
                    label: 'FFB Yield Curve f(x)',
                    data: yieldData,
                    borderColor: '#10b981',
                    borderWidth: 3,
                    backgroundColor: chartGradient,
                    fill: true,
                    tension: 0.3, // smooth curves
                    pointBackgroundColor: '#10b981',
                    pointHoverRadius: 8,
                    pointHoverBackgroundColor: '#3b82f6',
                    pointHoverBorderColor: '#ffffff',
                    pointHoverBorderWidth: 2
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: {
                    duration: 1500,
                    easing: 'easeOutQuart'
                },
                plugins: {
                    legend: {
                        display: true,
                        position: 'top',
                        labels: { font: { weight: 'bold' } }
                    },
                    tooltip: {
                        enabled: true,
                        callbacks: {
                            label: function(context) {
                                return ` Yield: ${context.parsed.y} T/Ha (Click to calculate f'(x))`;
                            }
                        }
                    }
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: 'Oil Palm Tree Age (Years)',
                            font: { weight: 'bold', size: 12 }
                        },
                        grid: { display: false }
                    },
                    y: {
                        title: {
                            display: true,
                            text: 'Yield (Tonnes / Hectare / Year)',
                            font: { weight: 'bold', size: 12 }
                        },
                        min: 0,
                        suggestedMax: 30
                    }
                },
                // Handle click events on data points
                onClick: (event, activeElements) => {
                    if (activeElements.length > 0) {
                        const index = activeElements[0].index;
                        const selectedYear = yearsData[index];
                        processCalculusAnalysis(selectedYear);
                    }
                }
            }
        });

        // 4. Calculus Analysis & Dynamic Layman Explanation Generation
        function processCalculusAnalysis(year) {
            const absoluteYield = calculateYield(year).toFixed(2);
            const derivativeVal = calculateDerivative(year).toFixed(2);
            const dVal = parseFloat(derivativeVal);

            // DOM Elements update
            document.getElementById('dashboard-prompt').classList.add('hidden');
            document.getElementById('dashboard-results').classList.remove('hidden');

            document.getElementById('result-age').innerText = `Age: ${year} Years Old`;
            document.getElementById('result-yield').innerText = `${absoluteYield} T/Ha`;
            document.getElementById('result-derivative').innerText = `${dVal > 0 ? '+' : ''}${derivativeVal}`;

            const dBox = document.getElementById('derivative-box');
            const dashBoard = document.getElementById('analysis-dashboard');
            let explanation = "";

            // Dynamic evaluation based on derivative sign and trend parameters
            if (year <= 4) {
                // Early growth/lag phase
                dBox.className = "p-3 rounded-lg border bg-blue-950/40 border-blue-700 text-blue-400";
                dashBoard.className = "bg-gray-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-4 border-blue-500";
                explanation = `At Year ${year}, the palm tree is in its youthful vegetative setup stage. The derivative is small but positive (${derivativeVal}). Yield is rising steadily as roots anchor and the canopy expands. Production velocity is building up.`;
            } 
            else if (year > 4 && year <= 11) {
                // Acceleration phase to inflection peak
                dBox.className = "p-3 rounded-lg border bg-emerald-950/40 border-emerald-700 text-emerald-400";
                dashBoard.className = "bg-gray-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-4 border-emerald-500";
                explanation = `At Year ${year}, the plantation is in its prime exponential growth curve. The derivative value reaches peak momentum around Year 12. Every year, you are gaining an extra ${derivativeVal} tonnes per hectare compared to the year before. Feed inputs should be maximized now to sustain this blistering acceleration.`;
            } 
            else if (year > 11 && year <= 13) {
                // Peak inflection window (f''(x) = 0 approx)
                dBox.className = "p-3 rounded-lg border bg-amber-950/40 border-amber-600 text-amber-400";
                dashBoard.className = "bg-gray-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-4 border-amber-500";
                explanation = `At Year ${year}, you have hit the biological inflection point (inflection zone). While total yield is near its ultimate high, the derivative shows that acceleration has flattened out to near zero. Growth speed is capping off.`;
            }
            else if (year > 13 && year < 24) {
                // Deceleration phase (f'(x) is still positive but decreasing velocity)
                dBox.className = "p-3 rounded-lg border bg-orange-950/40 border-orange-700 text-orange-400";
                dashBoard.className = "bg-gray-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-4 border-orange-500";
                explanation = `At Year ${year}, the trees are mature and large. Total absolute yield is high, but look at the derivative: it is shrinking. This means while you are still gaining yield, the rate of increase is slowing down drastically due to height-related harvest difficulties and older canopy metabolics.`;
            }
            else {
                // Senescence Phase / Negative Derivative
                dBox.className = "p-3 rounded-lg border bg-rose-950/40 border-rose-700 text-rose-400";
                dashBoard.className = "bg-gray-900 text-white p-6 rounded-xl shadow-md transition-all duration-300 border-l-4 border-rose-500";
                explanation = `At Year ${year}, the derivative has crossed into the negative zone (${derivativeVal}). Total biological yield is actively contracting each year due to severe tree aging and height limitations. To maintain national food security buffers, this is the optimal operational window to log and schedule systematic replanting.`;
            }

            document.getElementById('result-explanation').innerHTML = explanation;
        }
    </script>
